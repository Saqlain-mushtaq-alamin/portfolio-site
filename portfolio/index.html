class ARIA:
  def __init__(self):
    self.memory = ChromaDB()
    self.vision = OpenCV()
    self.voice = Whisper()
    self.llm = Ollama("llama3")

  async def think(self, query):
    ctx = await self.memory.recall()
    return await self.llm.reason(
      query, context=ctx
    )
